    31→
    32→function FadeIn({
    33→  children,
    34→  delay = 0,
    35→  direction = "up",
    36→  className = "",
    37→}: {
    38→  children: React.ReactNode;
    39→  delay?: number;
    40→  direction?: "up" | "down" | "left" | "right";
    41→  className?: string;
    42→}) {
    43→  const ref = useRef(null);
    44→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    45→
    46→  const dir = {
    47→    up: { y: 28, x: 0 },
    48→    down: { y: -28, x: 0 },
    49→    left: { y: 0, x: 28 },
    50→    right: { y: 0, x: -28 },
    51→  }[direction];
    52→
    53→  return (
    54→    <motion.div
    55→      ref={ref}
    56→      initial={{ opacity: 0, ...dir }}
    57→      animate={isInView ? { opacity: 1, y: 0, x: 0 } : {}}
    58→      transition={{ duration: 0.6, delay, ease: [0.22, 1, 0.36, 1] }}
    59→      className={className}
    60→    >
    61→      {children}
    62→    </motion.div>
    63→  );
    64→}

    66→function StaggerContainer({
    67→  children,
    68→  className = "",
    69→}: {
    70→  children: React.ReactNode;
    71→  className?: string;
    72→}) {
    73→  const ref = useRef(null);

    78→      ref={ref}
    79→      initial="hidden"
    80→      animate={isInView ? "visible" : "hidden"}
    81→      variants={{
    82→        visible: { transition: { staggerChildren: 0.08 } },
    83→        hidden: {},
    84→      }}
    85→      className={className}
    86→    >
    87→      {children}
    88→    </motion.div>
    89→  );
    90→}
