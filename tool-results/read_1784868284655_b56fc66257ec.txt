    32→    32→function FadeIn({
    33→    33→  children,
    34→    34→  delay = 0,
    35→    35→  direction = "up",
    36→    36→  className = "",
    37→    37→}: {
    38→    38→  children: React.ReactNode;
    39→    39→  delay?: number;
    40→    40→  direction?: "up" | "down" | "left" | "right";
    41→    41→  className?: string;
    42→    42→}) {
    43→    43→  const ref = useRef(null);
    44→    44→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    45→    45→
    46→    46→  const dir = {
    47→    47→    up: { y: 28, x: 0 },
    48→    48→    down: { y: -28, x: 0 },
    49→    49→    left: { y: 0, x: 28 },
    50→    50→    right: { y: 0, x: -28 },
    51→    51→  }[direction];
    52→    52→
    53→    53→  return (
    54→    54→    <motion.div
    55→    55→      ref={ref}
    56→    56→      initial={{ opacity: 0, ...dir }}
    57→    57→      animate={isInView ? { opacity: 1, y: 0, x: 0 } : {}}
    58→    58→      transition={{ duration: 0.6, delay, ease: [0.22, 1, 0.36, 1] }}
    59→    59→      className={className}
    60→    60→    >
    61→    61→      {children}
    62→    62→    </motion.div>
    63→    63→  );
    64→    64→}
